cmake_minimum_required(VERSION 3.13)
project(mhabit_gtk_title_bar_tests LANGUAGES CXX)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
add_executable(native_title_bar_test native_title_bar_test.cc ../native_title_bar.cc)
target_compile_features(native_title_bar_test PRIVATE cxx_std_14)
target_compile_options(native_title_bar_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(native_title_bar_test PRIVATE PkgConfig::GTK)
add_test(NAME native_title_bar COMMAND native_title_bar_test)
add_executable(native_window_persistence_test native_window_persistence_test.cc ../native_window_persistence.cc)
target_compile_features(native_window_persistence_test PRIVATE cxx_std_14)
target_compile_options(native_window_persistence_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(native_window_persistence_test PRIVATE PkgConfig::GTK)
add_test(NAME native_window_persistence COMMAND native_window_persistence_test)
